\section{Definitions and dependency graph}

\begin{definition}[Selector and shifted-prime set]\label{selector}
For integer $n$ put
\[
 Z(n)=\frac{\alpha\beta}{2}n(n-1)-\alpha n\lfloor\beta n\rfloor,
 \qquad Z_{\mathrm{br}}(n)=\alpha(n+1)\lfloor\beta(n+1)\rfloor.
\]
Here $\inf_{m\ge1}m\tn{m\alpha}>0$. The number $\beta\in(0,1)$
has increasing reduced prime-denominator approximants $a_j/r_j$
with $r_j\to\infty$,
$r_{j+1}\asymp\exp(\exp(\sqrt{\log r_j}))$ and
$0<\beta-a_j/r_j\ll r_{j+1}^{-1}$.
For an open arc $J\subset\T=\R/\Z$ with $0<|J|<1$,
$\mathcal B_{Z,J}=\{n\in\Z_{>0}:Z(n)\bmod 1\in J\}$.
The shifted-prime selectors are
$\mathcal P_{V,J}=\{p\in\PP:V(p-1)\bmod 1\in J\}$ for
$V=Z$ or $Z_{\mathrm{br}}$. Write
$\pi_{V,J}(N)=|\mathcal P_{V,J}\cap[N]|$.
\end{definition}

\begin{definition}[Rescaled counting and phase tests]\label{d:Cq}
Write $[M]=\{1,\ldots,M\}$, $P_0=0$, and fix integral polynomials
$P_i(0)=0$ of distinct increasing positive degrees with maximum $d$.
For a positive integer $q$, the shift $P_{i,q}(y)=P_i(qy)/q$ is integral.
A degree-$e$ polynomial with zero constant term has $(C,q)$-coefficients
if its leading coefficient is $a'_e q^{e-1}$ with $0<|a'_e|\le C$
and every lower nonconstant coefficient has absolute value at most $C$
times that of the leading coefficient. This class includes all $P_{i,q}$
for some fixed $C$.

For any such family $R_i$, $R_0=0$, and $1$-bounded spatial inputs
of finite support, the test count and polynomial-phase norm are
\[
 T_{N,M,v}(\mathbf f)
   =\frac{1}{NM}\sum_{x\in\Z}\sum_{y\in[M]}
        v(y)\prod_{i=0}^{k-1}f_i(x+R_i(y)),\qquad
 \norm{v}_{u^s[P]}
   =\sup_{\deg\phi<s}\abs{\E_{y\in P}v(y)\ee(-\phi(y))}.
\]
The tests use real polynomials, $P$ is a nonempty finite progression,
and $\ee(t)=e^{2\pi it}$. For a nonempty finite interval or progression
$I$, the full $U^s[I]$ norm takes the alternating
conjugation cube sum of the zero extension and divides the resulting
norm by that of $\one_I$. The normalized prime weight is
$\Lambda_q(y)=(\varphi(q)/q)\Lambda(qy+1)$, and
$F_q(y)=F(Z(qy))$ with $\mu_F=\int_\T F$.
\end{definition}

\begin{lemma}[6.1 and 4.3, horizontal and vertical cancellation]\label{p:heisenberg-phase}
\lean{NilBohr.horizontal_diophantine_bound_uniform_in_beta,NilBohr.heisenbergPhaseEstimates_of_quantitativeLeibman,NilBohr.scalar_heisenberg_log_saving,NilBohr.twisted_heisenberg_log_saving,NilBohr.centeredHeisenbergPhaseEstimate_of_obstructions}\leanok
\uses{selector}
Choose $r_j\le X^{1/100}<r_{j+1}$. For every fixed $L>0$,
$r_j/(\log X)^L\to\infty$. Uniformly for $1\le v\le X^{1/5000}$
with $r_j\nmid v$ and $0<\max(|k|,|l|)\le(\log X)^L$,
$\tn{v(k\alpha+l\beta)}\gg_\alpha X^{-1/20}$.

For $a\in\Z$, $1\le v\le X^{1/10000}$, $r_j\nmid v$ and
$K\ge X^{1/2}$, the Heisenberg orbit on $a+v[K]$
has arbitrary inverse-logarithmic equidistribution for Lipschitz tests
with polylogarithmic norm. Tests of zero vertical mean retain this saving
under any polynomial twist of fixed maximum degree.
In particular $F(Z(a+vt))-\mu_F$ has such cancellation.
\end{lemma}
\begin{proof}[Proof architecture]
\proves{p:heisenberg-phase}\uses{external-inputs,checked-floor,checked-twisted,checked-centered,checked-green-tao-adapter}\leanok
The horizontal bound compares $\beta$ to $a_j/r_j$ and uses bad
approximability for $k\ne0$. Apply quantitative Leibman to the orbit
with a circle carrying the polynomial test. A horizontal relation
must have nonzero circle coefficient $m$. Integer binomial coefficients
are periodic after division by $m$ on classes modulo $|m|d!$.
Absorb the horizontal phase into the cover induced by $D_{|m|}$
and use partial summation for the small remaining coefficients.
Cut off the horizontal chart boundary and use scalar equidistribution
to remove the cutoff.
\end{proof}
\noindent\textbf{Lean status.} The horizontal estimate is checked.
The scalar, twisted, and centered estimates use the scalar and product
criteria deduced from the general quantitative Leibman input.
Lemmas~\ref{checked-group-model} and~\ref{checked-green-tao-adapter}
record the verified metric, Haar, rational Mal'cev, and application steps.

\begin{lemma}[4.5, localized vertical multiplier]\label{l:vertical-mode-multiplier}
\uses{selector,d:Cq}
\lean{NilBohr.localized_vertical_multiplier,NilBohr.localized_vertical_multiplier_of_bernstein}\leanok
For $S\ge3$ and a finite nonempty integer interval $I$, let $\chi$ be
Lipschitz on $\T$ with support in a subinterval of $(0,1)$ of length
$<1/2$. Then for integers $a,v,h$,
\[
 \norm{f(t)\chi(\{\beta(a+vt)\})\ee(hZ(a+vt))}_{U^S[I]}
 \le \norm{\widehat\chi}_{\ell^1}\norm{f}_{U^S[I]}.
\]
\end{lemma}
\begin{proof}[Proof architecture]
\proves{l:vertical-mode-multiplier}\uses{checked-cube,checked-norm-foundations}\leanok
On any surviving cube the second differences of the floors are integers
of magnitude less than one. The floors are affine on the cube and the
vertical phase is quadratic. Its alternating sum vanishes, so the
phase leaves the localized cube sum unchanged. Fourier-expand the
cutoff and apply the triangle inequality and linear-phase invariance.
\end{proof}
\noindent\textbf{Lean status.} The multiplier inequality is checked
under the manuscript's hypotheses. The triangle inequality and the
absolute Fourier bound are proved in Lemma~\ref{checked-norm-foundations}.

\begin{theorem}[4.2, local generalized von Neumann bound]\label{t:main}
\lean{NilBohr.LocalGeneralizedVonNeumann,NilBohr.localGeneralizedVonNeumann_of_kmtt,NilBohr.localGeneralizedVonNeumann_twoInput,NilBohr.countingForm_twoInput_le}\leanok
\uses{d:Cq}
Fix $C\ge1$, $d$, and $k\ge2$. Set
$M=\lfloor(N/q^{d-1})^{1/d}\rfloor\ge1$ with $N\ge3$ and
$1\le q\le\exp((\log N)^{1/10})$. Let the shifted polynomials have
$(C,q)$-coefficients and distinct increasing positive degrees ending in $d$.
For $1$-bounded $v$ on $[M]$ and $1$-bounded $f_i$
supported in $[-CN,CN]$, there are constants
$\eta,A_1,A_2>0$ and $B\in\N$ depending only on $C,d,k$ such that
\[
 \abs{T_{N,M,v}(\mathbf f)}
 \ll_{C,d,k} N^{-\eta}+\rho+
   \left(\sup_{I,b,u}\norm{v}_{u^{d+1}[I\cap(u+q^b\Z)]}\right)^\eta
 \qquad (0<\rho\le1/2).
\]
Here $I\subset[M]$ has length at least $Mq^{-A_1}\rho^{A_2}$,
$0\le b\le B$, and the class meets $I$. For $k=2$ use $b=0$, $I=[M]$.
\end{theorem}
\begin{proof}[Proof architecture]
\proves{t:main}\uses{external-inputs,checked-local-phase}\leanok
Lemma~5.1 uses spatial Cauchy--Schwarz to remove $v$, then the
coefficient-uniform inverse theorem and local Plancherel to extract
piecewise linear phases on blocks of lengths $q^BV_i$ and classes
modulo $q^B$. The degree-uniform inverse interface lets Lean choose
$B$ using only $d$, as required in Lemma~5.1. Apply the extraction
to dual functions to insert one phase per
shifted input. Delete short parameter tails and pairs $(x,I)$ that
cross block boundaries. On a fixed interval and residue class the
product of phases has degree at most $d$. The residue contributions
are averaged using their relative sizes. The two-input case follows
from Fourier inversion in $x$.
\end{proof}
\noindent\textbf{Lean status.} The theorem is checked from the
KMTT inverse estimate. The proof bounds the discarded parameter tail
and spatial boundary pairs, then chooses the interval length and
converts the resulting test into the stated power-law estimate.

\begin{theorem}[3.2, smooth-modulus comparison]\label{t:relative-selector-comparison}
\lean{NilBohr.RelativeSelectorComparison,NilBohr.relativeSelectorComparison_of_obstructions,NilBohr.relativeSelectorComparison_of_centered_phase,NilBohr.relativeSelectorComparison_of_progression_estimates}\leanok
\uses{d:Cq,selector}
Let $q,w$ satisfy $w_0\le w\le(\log\widetilde Q_N)^{c_{\rm rel}}$,
$8\prod_{p\le w}p\mid q$, $P^+(q)\le w$ and $\log q\le w^2$,
where $\widetilde Q_N=\lceil\exp((\log N)^{c_0})\rceil$,
$0<c_0<1/11$. For fixed selector data and Lipschitz $F$,
there are $c_{\rm rel},\eta_{\rm sel}>0$ and $w_0\ge16$ such that,
for all sufficiently large $N$ and coefficient-uniform families as in
Theorem~4.2 at scale $M=\lfloor(N/q^{d-1})^{1/d}\rfloor$,
\[
 T_{N,M,\Lambda_qF_q}(\mathbf f)
 =\mu_F T_{N,M,1}(\mathbf f)+O(w^{-\eta_{\rm sel}})
\]
for all $1$-bounded inputs supported on $[N]$.
\end{theorem}
\begin{proof}[Proof architecture]
\proves{t:relative-selector-comparison}
\uses{t:main,p:heisenberg-phase,l:vertical-mode-multiplier,external-inputs,checked-selector-cutoffs}\leanok
Split the weight as $(\Lambda_q-1)F_q+(F_q-\mu_F)$ after centering.
Lemma~4.4 bounds the second summand by Theorem~4.2 and phase cancellation
on progressions of step $q^{b+1}$. Proposition~4.1 and the localized
multiplier bound the first summand by $O(\varepsilon^{-1}w^{-\eta_{\rm rel}})$
away from the chart boundary. There the polynomial tests use a
different norm from the prime-discrepancy estimate. Scalar
equidistribution and positivity control the boundary by
$O(\varepsilon+\varepsilon^{-1}w^{-\eta_{\rm rel}})$.
\end{proof}
\noindent\textbf{Lean status.} The uniform comparison is checked from
the KMTT prime and counting inputs, Banks's zero exclusion and the
quantitative Leibman theorem. The geometric cutoffs, norm estimates,
scalar and centered progression tests, and error absorption are checked.

\begin{theorem}[1.2, recurrence]\label{t:recurrence}
\lean{NilBohr.HeisenbergRecurrence,NilBohr.heisenbergRecurrence_of_analytic_inputs,NilBohr.heisenbergRecurrence_of_obstructions,NilBohr.heisenbergRecurrence_of_centered_phase,NilBohr.heisenbergRecurrence_of_relativeSelectorComparison}\leanok

such that for sufficiently large $N$ every $A\subset[N]$ of size at

\end{theorem}
\begin{proof}[Proof architecture]
\proves{t:recurrence}\uses{t:relative-selector-comparison,checked-weak-regularity,checked-density}\leanok

In the absence of selected configurations, the count comparison

step $q^B$ with density at least $(1+\eta)\delta$ and length at least

Set $q'=q^{B+1}$ and lift through the affine embedding. The growth
bounds allow enough increments to force density greater than one.
\end{proof}
\noindent\textbf{Lean status.} Proposition~3.1 and Theorem~1.2 are
checked from the uniform comparison in Theorem~3.2 and weak regularity

\begin{proposition}[6.2, prime-weighted phase bound]\label{p:prime-phase}
\lean{NilBohr.PrimeLocalPhase,NilBohr.primeLocalPhase_of_literature,NilBohr.heisenbergPrimeLocalPhase_of_literature}\leanok
\uses{selector,d:Cq}
For fixed $d\ge1$ and $A,B,K,L>0$, let $N$ be sufficiently large,
$1\le q\le(\log N)^L$, and $M=\lfloor(N/q^{d-1})^{1/d}\rfloor$.
If $I\subset[M]$, $|I|\ge M(\log N)^{-K}$, and $0\le b\le B$,
then for $P=I\cap(u+q^b\Z)$ one has
$\norm{\Lambda_q(F_q-\mu_F)}_{u^{d+1}[P]}\ll(\log N)^{-A}$
uniformly over $I,b,u$.
\end{proposition}
\begin{proof}[Proof architecture]
\proves{p:prime-phase}\uses{p:heisenberg-phase,l:vertical-mode-multiplier,external-inputs,checked-prime-cramer,checked-prime-seam,checked-fourier-restriction}\leanok
Transport to $m=qy+1$ with $X=qM+1$. A horizontal seam cutoff of
width $\eta=(\log X)^{-A-2}$ costs $O_F(\eta\log X)$ against $\Lambda$.
The full centered cutoff test has zero vertical mean and quotient
Lipschitz norm $O_F(1+\eta^{-1})$. For $\Lambda-\Lambda_{\Cramer}$,
partition on short horizontal chart intervals and apply the
localized multiplier uniformly in the vertical frequency.
Bernstein's absolute Fourier convergence then controls the full
test. The interval and residue Fourier expansions cost
$O(\log X)$ and one, respectively.

For the model itself, apply the dimension-one fundamental lemma
at level $\lfloor X^{1/20000}\rfloor$. The sieve error saves every
logarithmic power. Good divisors give long progressions to which
the full centered test applies. Exceptional divisors contribute
$O(\log X/r_j)$ before Euler normalization. Compare $\log X$
with $\log N$, and apply $\varphi(q)/q\le1$ after changing
back to $y$.
\end{proof}
\noindent\textbf{Lean status.} The external premises are the quoted
prime-model, zero-free and sieve estimates and quantitative Leibman.
The Lean constants and large-scale threshold are uniform in the
progression and the polynomial test.


\begin{theorem}[1.4 and 1.5, common pointwise limit]\label{t:pointwise}
\lean{NilBohr.PrimePointwiseFor,NilBohr.smoothErgodicComparison_of_prime_model_distinct_degrees,NilBohr.smoothErgodicComparison_of_literature,NilBohr.heisenberg_pointwise_of_prime_model,NilBohr.heisenberg_pointwise_of_analytic_inputs,NilBohr.heisenberg_pointwise_of_literature,NilBohr.bracket_pointwise_of_prime_model,NilBohr.bracket_pointwise_of_analytic_inputs,NilBohr.bracket_pointwise_of_literature}\leanok

\end{theorem}
\begin{proof}[Proof architecture]
\proves{t:pointwise}\uses{p:prime-phase,external-inputs,checked-ergodic,checked-orbit,checked-rotations,checked-prime-distribution}\leanok

fixed-family estimate after orbit transference. Ordering the degrees
permutes the polynomial and function indices together, preserving
both products. The resulting
$L^1$ difference is $O_A((\log H)^{-A})$ for every $A>0$.
Summability at $H_j=\lfloor\exp(j^{1/3})\rfloor$, a small-gap
estimate, removal of higher prime powers, and partial summation

\section{The separating family}

\begin{proposition}[A.1, local tests and global norms]\label{p:separating-family}
\lean{NilBohr.SeparatingFamilyClaims,NilBohr.appendixA_separating_family_of_quantitativeLeibman,NilBohr.separatingFamily_of_obstructions,NilBohr.appendixA_separating_family}\leanok
\uses{selector,d:Cq}
Put $G_q(y)=F(Z(qy))-\mu_F$ for nonconstant Lipschitz $F$.
There are smooth-modulus scales $q,M,N=qM^2$ with
$\log q\asymp(\log r_j)^2$. The global norm satisfies
$\norm{G_q}_{u^3[M]}\gg_F r_j^{-1}$, while
$\norm{G_q}_{u^3[P]}\ll_{A,B,F}(\log M)^{-A}$ for every
$P=I\cap(u+q^b\Z)$ with $I\subset[M]$, $0\le b\le B$ and
$|P|\ge(qM)^{1/2}$. For fixed $S\ge3$,
$\norm{G_q}_{U^S[M]}\gg_{F,S}1$.
\end{proposition}
\begin{proof}[Proof architecture]
\proves{p:separating-family}\uses{external-inputs,p:heisenberg-phase,l:vertical-mode-multiplier,checked-separating-finite,checked-separating-scales,checked-separating-local,checked-separating-quadratic,checked-separating-gowers}\leanok
Choose $w=\lfloor\log r_j\rfloor$,
$q=\mathcal L(w)^{\lfloor w^2/\log\mathcal L(w)\rfloor}$,
and $M=r_j\lfloor r_{j+1}^{1/4}/r_j\rfloor$. The growth of
$r_{j+1}$ makes the modulus and fixed powers of its step
subpowers of $M$. With $X=qM$, it also gives
$r_j\le X^{1/100}<r_{j+1}$. On $r_j\mid y$, the approximating
rational fixes the floor and yields a quadratic test. Fixed-shift
van der Corput gives equidistribution there. The nonzero Fourier
coefficient of $F$ survives Fourier extraction of that residue class.
On the tested local progressions, apply the centered estimate
with coprime step $q^{b+1}$. A localized vertical mode instead
has a nonzero correlation on the full interval, which gives the
full Gowers lower bound through Lemma~4.5.
\end{proof}
\noindent\textbf{Lean status.} Proposition~A.1 is checked on one
explicit sequence. The local upper bound and full Gowers lower bound use the
specializations of quantitative Leibman. The ordinary phase lower
bound follows from the Diophantine hypotheses, finite Fourier
extraction and the quadratic Weyl argument.

\begin{lemma}[Comparing the upper bounds]\label{checked-separating-comparison}
\lean{NilBohr.SeparatingCountComparison,NilBohr.appendixA_count_comparison,NilBohr.separatingFixedFamily_bounds_of_bounded,NilBohr.separating_fixedDilationBound_tendsto,NilBohr.repeatedDilationPhaseBound_one}\leanok
\uses{p:separating-family,t:main,external-inputs,checked-separating-finite}
The fixed-family bound loses a factor of $q^2$ under zero

The repeated bound also diverges for $K>1$, and at $K=1$ it

\end{lemma}
\begin{proof}[Proof architecture]
\proves{checked-separating-comparison}\leanok
The exact phase norm under dilation is $q^{-1}\norm{G_q}_{u^3[M]}$.
Combined with the global lower bound, the powers controlling the
two dilated expressions are $q^{2-1/K}r_j^{-1/K}$ and
$q^{1-1/K}r_j^{-1/K}$. For the direct bound use
$\rho=(\log M)^{-1}$, which makes every progression tested in
Theorem~4.2 long enough for Proposition~A.1.
\end{proof}
\noindent\textbf{Lean status.} The comparison is checked from
the inputs of Proposition~A.1, the KMTT inverse estimate and
Ter\'av\'ainen's fixed-family estimate. The divergence statements
concern the displayed upper-bound expressions. The direct
estimate gives the uniform decay of the counts.

\begin{proposition}[Introductory counterexample]\label{checked-intro-fast-obstruction}
\uses{selector}
\lean{NilBohr.intro_fast_approximation_counterexample}\leanok
There is an irrational $\beta\in(0,1)$ with strictly increasing prime
denominators $r_j$ and $0<\beta-a_j/r_j<\exp(-r_j^j)$.
Put $N_j=\lfloor\exp(r_j^j/6)\rfloor$. For each $\alpha\in\R$ there are sets
$A_j\subset[N_j]$ in which every $x,x+n,x+n^2$ with $n\ge1$ has
$Z(n)\bmod 1\notin(1/4,1/2)$. The same sets have density above
$(\log N_j)^{-\kappa}$ eventually for every fixed $\kappa>0$.
\end{proposition}
\begin{proof}[Proof architecture]
\proves{checked-intro-fast-obstruction}\leanok
Round each rational approximant upward on a sufficiently fine prime
grid. On $r_j\Z\cap[N_j]$ the accumulated slope error is less than one,
so the phase is quadratic on the configuration parameters.
A largest torus color cell has density at least $1/(578r_j)$
eventually and its configurations have $\tn{Z(n)}<1/8$.
The estimate $\log N_j\ge r_j^j/12$ eventually compares the cell
density to every fixed inverse power of $\log N_j$. The expository
blueprint gives the construction and limit estimates.
\end{proof}
